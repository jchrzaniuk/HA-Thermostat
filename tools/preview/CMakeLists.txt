cmake_minimum_required(VERSION 3.16)
project(onx_preview C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(LVGL "${ROOT}/managed_components/lvgl__lvgl")
file(GLOB_RECURSE LVGL_SRC "${LVGL}/src/*.c")
add_library(lvgl STATIC ${LVGL_SRC})
target_include_directories(lvgl PUBLIC "${LVGL}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE=1)
add_executable(preview "${CMAKE_CURRENT_BINARY_DIR}/preview.cpp" "${ROOT}/main/ui_assets.c")
target_include_directories(preview PRIVATE "${ROOT}/main" "${ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
if(APPLE)
 execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
 target_compile_options(preview PRIVATE "-isystem${SDK}/usr/include/c++/v1")
endif()
target_link_libraries(preview PRIVATE lvgl)
